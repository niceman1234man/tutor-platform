<div class="question">

  <ul>

  </ul>

</div>
🧠 Step 1: Parse HTML in Backend (Node.js)

Install:

npm install cheerio
Controller example:
import cheerio from "cheerio";

export const importExamFromHTML = async (req, res) => {
  try {
    const html = req.body.html; // send HTML string from frontend

    const $ = cheerio.load(html);
    const questions = [];

    $(".question").each((i, el) => {
      const question = $(el).find("p").text();

      const options = [];
      $(el).find("li").each((j, li) => {
        options.push($(li).text());
      });

      const correctAnswer = Number($(el).find(".answer").text());
      const explanation = $(el).find(".explanation").text();

      questions.push({
        question,
        options,
        correctAnswer,
        explanation,
      });
    });

    res.json({ questions });
  } catch (err) {
    res.status(500).json({ message: err.message });
  }
};
🧑‍💻 Step 2: Frontend (Upload HTML)

Add input:

<input type="file" accept=".html" onChange={handleHTMLUpload} />
Handle upload:
const handleHTMLUpload = async (e) => {
  const file = e.target.files[0];
  const text = await file.text();

  const res = await API.post("/admin/exams/import-html", {
    html: text,
  });

  setExam((prev) => ({
    ...prev,
    questions: res.data.questions,
  }));
};
